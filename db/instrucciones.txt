-- Instrucciones para crear tablas y datos en la base de datos "movies"

-- FORMA RECOMENDADA: desde la raíz del repo, con api/.env configurado, correr
--   npm run dev:setup
-- Crea la base si no existe y corre todos los scripts de esta carpeta. No necesita psql.
-- npm run dev:reset recarga todo desde cero; npm run db:drop borra solo lo que agrega el TP.

-- Lo que sigue es la forma manual, con psql.

-- La base de datos "movies" debe existir: crearla previamente usando pgadmin o la consola.

-- Abrir cmd or PowerShell (MacOS: terminal).
-- Cambiar al directorio donde están los scripts.
-- Abrir la consola de Postgres (pgsql) y ejecutar:

chcp 65001
set PGCLIENTENCODING=UTF8

-- Cerrar la consola y ejecutar:

psql --single-transaction -d movies -U postgres -f 00_load_all.sql

-- Después, crear las tablas de usuarios, los índices de personas y los datos de prueba:

psql -d movies -U postgres -f 13_users.sql
psql -d movies -U postgres -f 14_people_indexes.sql
psql -d movies -U postgres -f 15_seed_users.sql

-- Los usuarios de prueba (ana_cine, bruno_m, carla_r, diego_p) tienen la contraseña movieweb2026.

-- Para borrar solo lo que agrega el TP (usuarios e índices), sin tocar las películas:

psql -d movies -U postgres -f drop_tp_tables.sql
